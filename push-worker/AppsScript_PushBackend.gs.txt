var PUSH_WORKER_SECRET = 'REPLACE_WITH_A_LONG_RANDOM_STRING';
var PUSH_SHEET_NAME = 'PushSubscriptions';

// Routes the three push-related actions; returns null for everything else
// so your existing doPost handles it as normal.
function pushHandleAction_(action, data) {
  if (action === 'saveSubscription') return pushSaveSubscription_(data);
  if (action === 'removeSubscription') return pushRemoveSubscription_(data);
  if (action === 'fetchSubscriptionsForWorker') return pushFetchAllForWorker_(data);
  return null;
}

function getPushSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(PUSH_SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(PUSH_SHEET_NAME);
    sheet.appendRow(['username', 'endpoint', 'subscriptionJson', 'updatedAt']);
  }
  return sheet;
}

// Called by the app whenever a device turns push on. Keyed by endpoint, so
// re-subscribing the same device (or a token refresh) just updates the row
// instead of creating a duplicate.
function pushSaveSubscription_(data) {
  var username = data.username || 'default';
  var sub = data.subscription;
  if (!sub || !sub.endpoint) return { status: 'error', message: 'Missing subscription' };

  var sheet = getPushSheet_();
  var rows = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][1] === sub.endpoint) {
      sheet.getRange(i + 1, 1, 1, 4).setValues([[username, sub.endpoint, JSON.stringify(sub), new Date().toISOString()]]);
      return { status: 'success' };
    }
  }
  sheet.appendRow([username, sub.endpoint, JSON.stringify(sub), new Date().toISOString()]);
  return { status: 'success' };
}

// Called by the app when push is turned off on a device, and by the Worker
// itself when a push send comes back 404/410 (subscription is dead).
function pushRemoveSubscription_(data) {
  var endpoint = data.endpoint;
  if (!endpoint) return { status: 'error', message: 'Missing endpoint' };

  var sheet = getPushSheet_();
  var rows = sheet.getDataRange().getValues();
  for (var i = rows.length - 1; i >= 1; i--) {
    if (rows[i][1] === endpoint) { sheet.deleteRow(i + 1); break; }
  }
  return { status: 'success' };
}

// Called only by the Cloudflare Worker's scheduled job, gated by the shared
// secret — this is the one endpoint that hands back every profile's push
// subscriptions in one call, so it must not be open to the public internet.
function pushFetchAllForWorker_(data) {
  if (!PUSH_WORKER_SECRET || data.secret !== PUSH_WORKER_SECRET) {
    return { status: 'error', message: 'Forbidden' };
  }

  var sheet = getPushSheet_();
  var rows = sheet.getDataRange().getValues();
  var subscriptions = [];
  for (var i = 1; i < rows.length; i++) {
    if (!rows[i][1]) continue;
    try {
      subscriptions.push({ username: rows[i][0], subscription: JSON.parse(rows[i][2]) });
    } catch (e) { /* skip a corrupted row rather than fail the whole batch */ }
  }
  return { status: 'success', subscriptions: subscriptions };
}
